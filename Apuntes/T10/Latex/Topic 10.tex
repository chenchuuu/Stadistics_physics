	\chapter{Quantum Ideal Gas}
	
	\section{Non-interacting Ising model}
	
	\noindent $H = -J \sum_{i} s_i$ where $s_i = \pm 1$ (spin)
	
	\esp
	
	\noindent We must to find:
	
	\esp
	
	\noindent $Z$ \\
	\noindent $A$ \\
	\noindent $U = \langle H \rangle$ \\
	\noindent $\langle M \rangle = \langle \sum s_i \rangle / N$
	
	\esp
	
	\noindent \textbf{For interacting case}
	
	\esp
	
	At $T_c$ my particles are disordered. Below $T_c$, symmetry is broken (spontaneous magnetization).
	
	\esp
	
	$Z = (z_i)^N$ 
	
	\esp
	
	$z_i = \sum_{s_i = \pm 1} e^{-\beta H_i} = e^{\beta J} + e^{-\beta J} = 2\cosh(\beta J)$
	
	\esp
	
	$Z = (2\cosh(\beta J))^N$
	
	\esp
	
	$A = -kT \ln Z = -NkT \ln(2\cosh(\beta J))$
	
	\esp
	
	$U = \left( -\frac{\partial \ln Z}{\partial \beta} \right)_{N, V} = -\frac{\partial}{\partial \beta} (N \ln(2\cosh(\beta J))) = -N \frac{2\sinh(\beta J)}{2\cosh(\beta J)} \cdot J = -NJ \tanh(\beta J)$
	
	\esp
	
	\noindent \textbf{In other way:}
	
	\esp
	
	$U = \langle H \rangle = -J \sum_i \langle s_i \rangle = -NJ \tanh(\beta J)$ \\
	$\langle s_i \rangle = 1 \cdot P(s_i=1) - 1 \cdot P(s_i=-1) = \frac{2\sinh(\beta J)}{2\cosh(\beta J)} = \tanh(\beta J)$
	
	\esp
	
	where $P(s_i) = \frac{e^{-\beta J s_i}}{2\cosh(\beta J)}$ \\
	$P(1) = \frac{e^{\beta J}}{2\cosh(\beta J)}$ \\
	$P(-1) = \frac{e^{-\beta J}}{2\cosh(\beta J)}$
	
	\section{Ideal Quantum Gas}
	
	System made of $N$ identical particles:
	\begin{equation}
		\Psi(r_1, r_2, \dots, r_N, t)
	\end{equation}
	
	\esp
	
	Its evolution is given by:
	\begin{equation}
		i\hbar \frac{\partial \Psi}{\partial t} = H\Psi \quad ; \quad H = \sum_{i=1}^N \frac{p_i^2}{2m} + V(r_1, r_2, \dots, r_N)
	\end{equation}
	
	\esp
	
	$P_{ij}$ is the \textbf{permutation operator} which exchanges the labels of particles $i$ and $j$:
	\begin{equation}
		P_{ij} \Psi(\dots, r_i, \dots, r_j, \dots, t) = \Psi(\dots, r_j, \dots, r_i, \dots, t)
	\end{equation}
	
	\esp
	
	$[H, P_{ij}] = 0 \quad \forall i, j \quad (i \neq j)$
	
	\esp
	
	If $H\phi_n = E_n\phi_n$:
	\begin{equation}
		\phi(r_1, \dots, r_N, t) = \sum a_n \phi_n(r_1, \dots, r_N) \quad ; \quad a_n = \langle \phi_n | \Psi \rangle
	\end{equation}
	
	\esp
	
	Now we apply $P_{ij}$ to $\Psi$ and then $H$:
	\begin{align*}
		H P_{ij} \Psi(\dots, r_i, \dots) &\to \\
		H \Psi(\dots, r_j, \dots, r_i, \dots, t)\\ 
		&= \sum a_n E_n \phi_n(\dots, r_j, \dots, r_i, \dots, t) = \sum a_n E_n \underbrace{P_{ij}}_{\text{scalars}} \phi_n(\dots, r_i, \dots, r_j, \dots, t) \\
		&= P_{ij} \sum a_n E_n \phi_n(\dots, r_i, \dots, r_j, \dots, t) = P_{ij} H \Psi \\
		&\to \text{We have proved that } [H, P_{ij}] = 0 \quad \forall i, j \quad (i \neq j)
	\end{align*}
	
	\esp
	
	Commuting observables have a complete set of common eigenfunctions:
	\begin{equation}
		P_{ij} H \phi = P_{ij} E \phi = E (P_{ij} \phi) \xrightarrow{[H, P_{ij}]=0} H (P_{ij} \phi)
	\end{equation}
	This implies $P_{ij} \phi$ is also an eigenfunction of $H$ with the eigenvalue $E$.
	
	\esp
	
	$P_{ij} \phi = \Theta \phi$
	
	\esp
	
	\textbf{Value of $\Theta$?}
	\begin{align*}
		P_{ij} \phi(\dots, r_i, \dots, r_j, \dots, t) &= \phi(\dots, r_j, \dots, r_i, \dots, t) \\
		P_{ij}^2 \phi(\dots, r_i, \dots, r_j, \dots, t) &= \phi(\dots, r_i, \dots, r_j, \dots, t) = I \phi
	\end{align*}
	Then $P_{ij}^2 = I$.
	
	\esp
	
	As $P_{ij} \phi = \Theta \phi$:
	\begin{align*}
		P_{ij} \phi(\dots, r_i, \dots, r_j, \dots, t) &= \Theta \phi(\dots, r_i, \dots, r_j, \dots, t) \\
		P_{ij}^2 \phi(\dots, r_i, \dots, r_j, \dots, t) &= \Theta^2 \phi(\dots, r_i, \dots, r_j, \dots, t)
	\end{align*}
	As $P_{ij}^2 = I$, then $\Theta^2 = 1 \implies \Theta = \pm 1$.
	
	\esp
	
	\begin{itemize}
		\item \textbf{Bosons:} $\Theta = +1 \to$ \textit{Symmetric}
		\item \textbf{Fermions:} $\Theta = -1 \to$ \textit{Antisymmetric}
	\end{itemize}
	
	\esp
	
	\textbf{What happens if two particles are in the same position ($r_i = r_j$) and I exchange them with $P_{ij}$?}
	
	\esp
	
	\textbf{First case (Fermions):} $P_{ij} \phi(\dots, r_i, \dots, r_i, \dots, t) = -\phi(\dots, r_i, \dots, r_i, \dots, t)$ \\
	$\implies \phi(\dots, r_i, \dots, r_i, \dots, t) = 0 \quad \to$ It is \textbf{forbidden} that two particles occupy the same position. These particles are known as \textbf{Fermions} (half-integer spin).
	
	\esp
	
	\textbf{Second case (Bosons):} $P_{ij} \phi(\dots, r_i, \dots, r_i, \dots, t) = \phi(\dots, r_i, \dots, r_i, \dots, t)$ \\
	$\implies \phi(\dots, r_i, \dots, r_i, \dots, t) = \phi(\dots, r_i, \dots, r_i, \dots, t) \quad \to$ It is \textbf{not forbidden} that two particles occupy the same position. These particles are known as \textbf{Bosons} (integer spin).
	
	\esp
	
	\textbf{Examples:}
	
	\esp
	
	\begin{table}[h]
		\centering
		\begin{tabular}{ll}
			\textbf{Fermions} & \textbf{Bosons} \\
			\textcolor{red}{Electron} & \textcolor{red}{Quark} \\
			\textcolor{red}{Proton} & \textcolor{red}{Gluon} \\
			\textcolor{red}{Neutron} & \textcolor{red}{$\alpha$ particle} \\
			$\vdots$ & $\vdots$ \\
			$\phi$ is antisymmetric ($\Theta = -1$) & $\phi$ is symmetric ($\Theta = +1$)
		\end{tabular}
	\end{table}
	
	\esp
	
	\textbf{Ideal $\to H = \sum_i^N H_i$}
	
	\esp
	
	$\sum_i n_i = N$ \\
	$\sum_i n_i \epsilon_i = E$
	
	\esp
	
	\textbf{Fermions vs Bosons occupancy:}
	
	\esp
	
	\begin{itemize}
		\item \textbf{Fermions:} $n_j \in \{0, 1\}$ (Pauli exclusion principle).
		\item \textbf{Bosons:} $n_j \in \{0, 1, 2, \dots, N\}$ (No restriction).
	\end{itemize}
	
	\esp
	
	\begin{equation}
		Z(T, V) = \sum_{\{n_i\}} e^{-\beta \sum n_i \epsilon_i} = \sum_{\{n_i\}} e^{-\beta E(\{n_i\})} \delta_{N, \sum n_i}
	\end{equation}
	$\to$ \textit{This restriction ($\sum n_i = N$) does not allow us to perform the sum easily.}
	
	\esp
	
	\textbf{Boltzmann approximation} $\to z_N = \sum_{\{n_i\}} \frac{n_1! n_2! \dots n_i!}{N!} e^{-\beta \sum n_i \epsilon_i} \approx \frac{1}{N!} \sum_{\{n_i\}} e^{-\beta \sum n_i \epsilon_i}$
	
	\esp
	
	\textbf{Ensembles Comparison:}
	
	\esp
	
	\begin{itemize}
		\item \textbf{Microcanonical:} $(E, V, N)$ fixed. Isolated system.
		\item \textbf{Canonical:} $(\langle E \rangle, V, N)$ fixed. Thermal contact with reservoir at $T$.
		\item \textbf{Grand Canonical (Macrocanonical):} $(\langle E \rangle, V, \langle N \rangle)$ fixed. Thermal and particle contact $(\mu, T)$.
	\end{itemize}
	
	\esp
	
	\noindent \textbf{Macrocanonical ensemble}
	
	\esp
	
	$\Xi(\mu, T, V) = \sum_{N=0}^{\infty} e^{\beta \mu N} Z_N(T, V) \quad \to$ \textit{similar to } $\int e^{tx} p(x) dx \to$ \textit{moment generating function}
	
	\esp
	
	\begin{align*}
		\Xi(\mu, T, V) &= \sum_{N=0}^{\infty} e^{\beta \mu N} \sum_{\{n_i\}} e^{-\beta \sum n_i \epsilon_i} \delta_{N, \sum n_i} = \sum_{\{n_i\}} e^{\beta \mu \sum n_i} \sum_{N=0}^{\infty} e^{\beta \mu N} \delta_{N, \sum n_i} \\
		&= \sum_{\{n_i\}} e^{-\beta \sum n_i \epsilon_i} e^{\beta \mu \sum n_i} = \sum_{\{n_i\}} e^{\beta \sum (\mu - \epsilon_i) n_i} = \sum_{\{n_i\}} \prod_i e^{\beta (\mu - \epsilon_i) n_i} \\
		&= \prod_i \sum_{\{n_i\}} e^{\beta (\mu - \epsilon_i) n_i}
	\end{align*}
	
	\esp
	
	\noindent \textbf{Fermions}
	
	\esp
	
	$n_i = 0, 1 \quad \forall i$
	
	\esp
	
	\noindent \textbf{then}
	
	\esp
	
	\begin{equation}
		\Xi(\mu, T, V) = \prod_i \left( 1 + e^{\beta (\mu - \epsilon_i)} \right)
	\end{equation}
	\textit{where 1 is when } $n_i=0$ \textit{ and the exponential is when } $n_i=1$.
	
	\esp
	
	\noindent \textbf{Bosons}
	
	\esp
	
	$n_i = 0, 1, \dots, \infty \quad \forall i$
	
	\esp
	
	\noindent \textbf{then}
	
	\esp
	
	\begin{equation}
		\Xi(\mu, T, V) = \prod_i \sum_{n_i=0}^{\infty} \left( e^{\beta (\mu - \epsilon_i)} \right)^{n_i} = \prod_i \left( 1 - e^{\beta (\mu - \epsilon_i)} \right)^{-1}
	\end{equation}
	\textit{using } $\sum x^n = \frac{1}{1-x}$ \textit{ for } $|x| < 1$.
	
	\esp
	
	\noindent \textbf{I can write generally this:}
	
	\esp
	
	\begin{equation}
		\Xi(\mu, T, V) = \prod_i \left( 1 - \Theta e^{\beta (\mu - \epsilon_i)} \right)^{-1/\Theta}
	\end{equation}
	
	\esp
	
	\textbf{where } $\Theta = \begin{cases} +1 & \text{for bosons} \\ -1 & \text{for fermions} \end{cases}$

	\esp
	
	\begin{table}[h]
		\centering
		\begin{tabular}{ll}
			\textbf{Canonical ensemble} & \textbf{Macrocanonical ensemble} \\
			$A = -kT \ln Z$ & $J = -kT \ln \Xi$ \\
			$A = U - TS$ & $J = U - TS - \mu N$
		\end{tabular}
	\end{table}
	
	\esp
	
	From the macrocanonical ensemble we are going to use:
	\begin{itemize}
		\item $J = -kT \ln \Xi$
		\item $PV = -J$
		\item $\langle N \rangle = - \left( \frac{\partial J}{\partial \mu} \right)_{T, V}$
	\end{itemize}
	
	\esp
	
	\textbf{From what we know:}
	\begin{equation}
		J = -kT \ln \Xi = -kT \ln \left[ \prod_i (1 - \Theta e^{\beta(\mu - \epsilon_i)})^{-1/\Theta} \right] = \frac{kT}{\Theta} \sum_i \ln(1 - \Theta e^{\beta(\mu - \epsilon_i)})
	\end{equation}
	
	\esp
	
	\begin{center}
		\textbf{For Bosons $\mu < 0$ \quad For Fermions $\mu$ can be any value}
	\end{center}
	
	\esp
	
	\begin{equation}
		\langle N \rangle = - \left( \frac{\partial J}{\partial \mu} \right)_{T, V} = -\frac{kT}{\Theta} \sum_i \frac{-\Theta \beta e^{\beta(\mu - \epsilon_i)}}{1 - \Theta e^{\beta(\mu - \epsilon_i)}} = \sum_i \frac{1}{e^{\beta(\epsilon_i - \mu)} - \Theta} \to \text{this remembers us the restriction } \sum n_i = N
	\end{equation}
	$\implies \langle N \rangle = \langle \sum n_i \rangle = \sum \langle n_i \rangle$
	
	\esp
	
	\textbf{So we can know the average of $n_i$ for each energy level $\epsilon_i$:}
	\begin{equation}
		\langle n_i \rangle = \frac{1}{e^{\beta(\epsilon_i - \mu)} - \Theta}
	\end{equation}
	\textit{Average occupation number of level $i$:}
	
	\esp
	
	\begin{itemize}
		\item \textbf{Fermi-Dirac Statistics (Fermions):} $\langle n_i \rangle = \frac{1}{e^{\beta(\epsilon_i - \mu)} + 1}$
		\item \textbf{Bose-Einstein Statistics (Bosons):} $\langle n_i \rangle = \frac{1}{e^{\beta(\epsilon_i - \mu)} - 1}$
	\end{itemize}
	
	\esp
	
	
	\esp
	
	\begin{equation}
		J = \frac{kT}{\Theta} \sum_i \ln(1 - \Theta e^{\beta(\mu - \epsilon_i)})
	\end{equation}
	
	\esp
	
	\begin{equation}
		\lim_{\Theta \to 0} J = \lim_{\Theta \to 0} -kT \sum_i \ln(1 - \Theta e^{\beta(\mu - \epsilon_i)})^{1/\Theta} = -kT \sum_i e^{\beta(\mu - \epsilon_i)}
	\end{equation}
	\textit{Using } $e^x = \lim_{n \to \infty} (1 + \frac{x}{n})^n$
	
	\esp
	
	\textbf{Remember that} $z_{MB} = \frac{1}{N!} (z_{MB})^N \quad \text{so} \quad \Xi(\mu, T, V) = \sum_{N=0}^{\infty} e^{\beta \mu N} z_{MB} = \sum_{N=0}^{\infty} \lambda^N \frac{z_1^N}{N!} = e^{\lambda z_1}$
	\textit{where } $\lambda$ \textit{ is the fugacity: } $\lambda = e^{\beta \mu}$
	
	\esp
	
	\noindent we reached the same result as the boltzmann gas approximation
	
	\esp
	
	$J = -kT \ln \Xi = -\lambda z_1 kT = -kT e^{\beta \mu} \sum_i e^{-\beta \epsilon_i} = -kT \sum_i e^{\beta(\mu - \epsilon_i)}$
	
	\esp
	
	\textbf{Then } $J(\mu, T, V) = \frac{kT}{\Theta} \sum_i \ln(1 - \Theta e^{\beta(\mu - \epsilon_i)})$ 
	$\begin{cases} \Theta = 1 & \text{Bosons} \\ \Theta = -1 & \text{Fermions} \\ \Theta \to 0 & \text{MB} \end{cases} \quad \to$ \textit{this unified the three statistics.}
	
	\esp
	
	$\langle n_i \rangle = \frac{1}{e^{\beta(\epsilon_i - \mu)} - \Theta}$
	
	\esp
	
	
	\esp
	
	\textbf{Boltzmann approximation} $\to$ We see that bosons and fermions converge to the same behavior.
	In this limit: $\langle n_i \rangle = e^{\beta(\mu - \epsilon_i)} \ll 1$. In 3D $\to \rho \Lambda^3 \ll 1$.
	
	\esp
	
	\textbf{Let's demonstrate that the Boltzmann approximation (which means $\langle n_i \rangle = e^{\beta(\mu - \epsilon_i)} \ll 1$) corresponds to considering $\rho \Lambda^3 \ll 1$}
	
	\esp
	
	For free particle in 3D: $z_1 = \frac{V}{\Lambda^3}$, then $Z = \frac{z_1^N}{N!}$ (where $L \gg \Lambda$) \\
	$\to$ \textit{as calculated before for the Boltzmann gas.}
	
	\esp
	
	$A(N, T, V) = -kT \ln Z = -kT \ln \left( \frac{V^N}{\Lambda^{3N} N!} \right)$
	
	\esp
	
	$\mu = \left( \frac{\partial A}{\partial N} \right)_{T, V} \to$ \textit{The chemical potential is the cost of Free Energy of varying $N$.}
	
	\esp
	
	A system always wants to minimize the Helmholtz Free Energy:
	$A = U - TS \begin{cases} \text{sometimes we just need to minimize } U \to \text{Example: } T=0 \\ \text{sometimes we just need to maximize } S \to \text{Example: fixed } U \end{cases}$
	
	\esp
	
	\textbf{What does $\mu$ mean to $A$?}
	
	\esp
	
	\begin{itemize}
		\item When $\mu < 0$, this means that as we increase $N$, we get a lower $A$. In this situation, the system wants more $N$ in order to decrease $A$.
		\item When $\mu > 0$, this means that as we decrease $N$, we get higher $A$. Then the system wants to reduce $N$ to decrease $A$.
	\end{itemize}
	
	\esp
	
	
	\esp
	
	\textbf{Boltzmann approximation:}
	$e^{\beta(\mu - \epsilon)} \ll 1 \implies \beta(\mu - \epsilon) \ll 0 \implies \mu\beta - \epsilon\beta \ll 0 \xrightarrow{\epsilon \to 0} \ln(\rho \Lambda^3) < 0 \implies \rho \Lambda^3 \ll 1$
	\textit{(If it's satisfied for $\epsilon=0$, the most unfavorable case, it's satisfied for any $\epsilon$)}.
	
	\esp
	
	Thus, the Boltzmann approximation is equivalent to considering $\rho \ll 1$ and/or $T \gg 1$.
	
	\esp
	
	\textbf{Thermodynamic properties of Bosons and Fermions}
	
	\esp
	
	$J(\mu, T, V) = \frac{kT}{\Theta} \sum_i \ln(1 - \Theta e^{\beta(\mu - \epsilon_i)})$
	
	\esp
	
	We are going to consider \textbf{free particles in a 3D box}. What is the spectrum in this case?
	
	\esp
	
	$H \psi_{n_x, n_y, n_z} = \epsilon_{n_x, n_y, n_z} \psi_{n_x, n_y, n_z}$ \\
	$H = \sum_{i=1}^N -\frac{\hbar^2}{2m} \nabla_i^2$ \\
	$\epsilon_{n_x, n_y, n_z} = \epsilon_{nx} + \epsilon_{ny} + \epsilon_{nz}$ \\
	$\epsilon_{nx} = \frac{\hbar^2}{2m} \frac{\pi^2}{L^2} n_x^2 \quad \text{where } n = x, y, z$ \\
	$\psi_{n_x, n_y, n_z} = \left( \frac{2}{L} \right)^{3/2} \sin\left( \frac{n_x \pi x}{L} \right) \sin\left( \frac{n_y \pi y}{L} \right) \sin\left( \frac{n_z \pi z}{L} \right)$
	
	\esp
	
	As $L \to \infty$, we are going to perform the sum over $n_x, n_y, n_z$ instead of $\epsilon$:
	$J(\mu, T, V) = \frac{kT}{\Theta} \sum_{n_x=1}^\infty \sum_{n_y=1}^\infty \sum_{n_z=1}^\infty \ln(1 - \Theta e^{\beta(\mu - \epsilon_{n_x, n_y, n_z})})$
	
	\esp
	
	$\xrightarrow{\text{change of variable}} p_{nx} = \frac{\hbar \pi n_x}{L} \to \epsilon_{n_x, n_y, n_z} = \frac{p_{nx}^2 + p_{ny}^2 + p_{nz}^2}{2m}$
	
	\esp
	
	$J(\mu, T, V) = \frac{kT}{\theta} \sum_{p_x = \frac{h}{2L}} \sum_{p_y = \frac{h}{2L}} \sum_{p_z = \frac{h}{2L}} \ln (1 - \theta e^{\beta (\mu - \frac{(p_x^2 + p_y^2 + p_z^2)}{2m})})$
	
	\esp
	
	\noindent when $L \gg 1$
	
	\esp
	\begin{equation*}
		\sum_{p_x} \sum_{p_y} \sum_{p_z} \rightarrow \frac{g L^3}{h^3} \int_{0}^{\infty} dp_x \int_{0}^{\infty} dp_y \int_{0}^{\infty} dp_z f(p) \rightarrow \frac{g L^3}{2^3 h^3} \int_{-\infty}^{\infty} dp_x \int_{-\infty}^{\infty} dp_y \int_{-\infty}^{\infty} dp_z f(p) 
	\end{equation*}
	\noindent where $f(p) = f(-p)$ is an even function.
	
	\esp
	\noindent Using $dp_x dp_y dp_z = 4\pi p^2 dp$:
	\begin{equation*}
		\rightarrow \frac{4\pi V}{h^3} \int_{0}^{\infty} p^2 f(p) dp = \int_{0}^{\infty} D(p) f(p) dp \quad \text{where} \quad \boxed{D(p) = \frac{V}{h^3} 4\pi p^2} \text{ density of states}
	\end{equation*}
	We are going to see this many times. $D(p)dp \rightarrow$ number of states in $[p, p+dp]$.
	
	\esp
	Since we have a free particle: $\varepsilon = \frac{p^2}{2m}$, I can express $D(p)$ as $D(\varepsilon)$. $D(p)dp = D(\varepsilon)d\varepsilon$ must be the same, the number of states couldn't change when I change the variable.
	
	\esp
	\begin{equation*}
		D(\varepsilon)d\varepsilon = D(p)dp \rightarrow D(\varepsilon)\frac{d\varepsilon}{dp} = D(p) \rightarrow D(\varepsilon)\frac{p}{m} = \frac{V}{h^3} 4\pi 2m\varepsilon \rightarrow D(\varepsilon)\frac{\sqrt{2m\varepsilon}}{m} = \frac{V}{h^3} 4\pi 2m\varepsilon
	\end{equation*}
	\begin{equation*}
		\rightarrow D(\varepsilon) = \frac{V}{h^3} \frac{m}{\sqrt{2m\varepsilon}} 4\pi 2m\varepsilon = \frac{V}{h^3} 4\pi m \sqrt{2m\varepsilon} = \frac{V}{h^3} 4\pi m^{3/2} \sqrt{2\varepsilon}
	\end{equation*}
	
	\esp
	$\hookrightarrow D(\varepsilon) = D(p)\frac{dp}{d\varepsilon} = \dots$ same result.
	
	\esp
	
	$J(\mu, T, V) = \frac{kT}{\theta} \frac{V}{h^3} 4\pi \int_{0}^{\infty} p^2 \ln(1 - \theta e^{\beta(\mu - p^2/2m)}) dp = \frac{kT}{\theta} \int_{0}^{\infty} D(p) \ln(1 - \theta e^{\beta(\mu - p^2/2m)}) dp = \frac{kT}{\theta} \int_{0}^{\infty} D(\varepsilon) \ln(1 - \theta e^{\beta(\mu - \varepsilon)}) d\varepsilon$
	
	\esp
	As factor of energy $\rightarrow$ we changed $D(p)dp = D(\varepsilon)d\varepsilon$ so we have not quantum constant.
	
	\esp
